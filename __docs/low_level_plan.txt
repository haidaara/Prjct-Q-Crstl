```python
class LegalFlipEngine:
    def __init__(self, vertex_classifier, energy_model):
        self.classifier = vertex_classifier
        self.energy_model = energy_model
        self.flip_stats = {"proposed": 0, "accepted": 0, "rejected": 0}
    
    def find_flippable_hexagons(self, tiling_data) -> List[List[int]]:
        """
        Returns: List of clusters, where each cluster is [tile_id1, tile_id2, tile_id3]
        """
        flippable_clusters = []
        
        for tile in tiling_data["tiles"]:
            if not self._is_eligible_for_flip(tile):
                continue
                
            cluster_ids = self._detect_hexagon_pattern(tile["id"], tiling_data)
            if cluster_ids and self._validate_flip_legality(cluster_ids, tiling_data):
                flippable_clusters.append(cluster_ids)
        
        return flippable_clusters
    
    def _is_eligible_for_flip(self, tile) -> bool:
        """Check using EXISTING obstacle fields"""
        return (not tile.get("removed", False) and 
                not tile.get("immobile", False) and
                tile.get("flippable", True))
    
    def propose_flip(self, cluster_ids: List[int], tiling_data) -> Tuple[Dict, Dict]:
        """
        CRITICAL FIX: Return BOTH new state AND undo info
        Does NOT mutate original tiling_data
        """
        # 1. Create deep copy for proposal
        new_tiling = copy.deepcopy(tiling_data)
        undo_info = self._capture_current_state(cluster_ids, tiling_data)
        
        # 2. Apply combinatorial transformation
        self._apply_hexagon_transformation(cluster_ids, new_tiling)
        
        # 3. Update physics fields for affected tiles
        affected_tiles = self._get_affected_tile_neighborhood(cluster_ids, new_tiling)
        for tile_id in affected_tiles:
            self._update_tile_physics(tile_id, new_tiling)
        
        return new_tiling, undo_info
    
    def execute_flip(self, cluster_ids: List[int], tiling_data) -> None:
        """
        CRITICAL FIX: Mutate tiling_data in-place after acceptance
        """
        self._apply_hexagon_transformation(cluster_ids, tiling_data)
        
        # Update physics fields in original tiling
        affected_tiles = self._get_affected_tile_neighborhood(cluster_ids, tiling_data)
        for tile_id in affected_tiles:
            self._update_tile_physics(tile_id, tiling_data)
    
    def undo_flip(self, cluster_ids: List[int], undo_info: Dict, tiling_data) -> None:
        """Restore original state for rejection"""
        for tile_id, original_state in undo_info.items():
            tile = tiling_data["tiles"][tile_id]
            tile.update(original_state)  # Restore all fields
    
    def _capture_current_state(self, cluster_ids: List[int], tiling_data) -> Dict:
        """Capture current state of cluster and neighbors for undo"""
        affected_tiles = self._get_affected_tile_neighborhood(cluster_ids, tiling_data)
        undo_info = {}
        
        for tile_id in affected_tiles:
            tile = tiling_data["tiles"][tile_id]
            undo_info[tile_id] = {
                "type": tile["type"],
                "vertex_class": tile["vertex_class"],
                "local_energy": tile["local_energy"],
                "neighbors": tile["neighbors"][:]  # Copy list
            }
        
        return undo_info
    
    def _update_tile_physics(self, tile_id: int, tiling_data) -> None:
        """Update EXISTING physics fields using current methods"""
        tile = tiling_data["tiles"][tile_id]
        
        # Use EXISTING classifier and energy model
        tile["vertex_class"] = self.classifier.classify_vertex_environment(tile_id, tiling_data)
        tile["local_energy"] = self.energy_model.compute_local_energy(tile_id, tiling_data)
```

```python
class LocalMCEngine:
    def __init__(self, temperature=0.3, energy_model=None, flip_engine=None):
        self.temperature = temperature
        self.energy_model = energy_model
        self.flip_engine = flip_engine
        
        # Track global energy to avoid recomputation
        self.current_energy = 0.0
        self.metrics = {
            "energy_history": [],
            "acceptance_rates": [],
            "computation_speeds": []
        }
    
    def initialize_energy(self, tiling_data) -> None:
        """Compute initial total energy once"""
        self.current_energy = self.energy_model.compute_total_energy(tiling_data)
        self.metrics["energy_history"].append(self.current_energy)
    
    def run_mc_step(self, tiling_data) -> bool:
        """
        Single MC step with PROPER state management
        Returns: True if accepted, False if rejected
        """
        # 1. Find legal flips (returns list of tile ID clusters)
        flippable = self.flip_engine.find_flippable_hexagons(tiling_data)
        if not flippable:
            return False
        
        # 2. Random proposal
        cluster_ids = random.choice(flippable)
        
        # 3. Compute LOCAL energy change (O(1))
        affected_tiles = self.flip_engine._get_affected_tile_neighborhood(cluster_ids, tiling_data)
        current_local_energy = self._compute_local_energy_sum(affected_tiles, tiling_data)
        
        # 4. PROPOSAL: Get new state WITHOUT mutation
        proposed_tiling, undo_info = self.flip_engine.propose_flip(cluster_ids, tiling_data)
        new_local_energy = self._compute_local_energy_sum(affected_tiles, proposed_tiling)
        
        delta_energy = new_local_energy - current_local_energy
        
        # 5. Metropolis acceptance
        if self._metropolis_accept(delta_energy):
            # ACCEPTED: Execute flip on original tiling
            self.flip_engine.execute_flip(cluster_ids, tiling_data)
            
            # Update global energy (O(1) update)
            self.current_energy += delta_energy
            self.metrics["energy_history"].append(self.current_energy)
            self.flip_engine.flip_stats["accepted"] += 1
            return True
        else:
            # REJECTED: Original tiling remains unchanged
            self.flip_engine.flip_stats["rejected"] += 1
            return False
    
    def run_mc_sweep(self, tiling_data, steps: int = 500) -> None:
        """Run multiple MC steps with proper energy tracking"""
        acceptance_count = 0
        
        for step in range(steps):
            accepted = self.run_mc_step(tiling_data)
            if accepted:
                acceptance_count += 1
            
            # Progress monitoring
            if step % 100 == 0:
                print(f"MC Step {step}: E={self.current_energy:.2f}, " 
                      f"Acceptance={acceptance_count/(step+1):.1%}")
        
        # Update metrics
        acceptance_rate = acceptance_count / steps
        self.metrics["acceptance_rates"].append(acceptance_rate)
    
    def _compute_local_energy_sum(self, tile_ids: List[int], tiling_data) -> float:
        """Compute sum of local energies for specific tiles"""
        return sum(self.energy_model.compute_local_energy(tid, tiling_data) 
                   for tid in tile_ids)
    
    def _metropolis_accept(self, delta_energy: float) -> bool:
        """Standard Metropolis criterion"""
        if delta_energy <= 0:
            return True
        return random.random() < math.exp(-delta_energy / self.temperature)
```

```python
class GrowthFrontSimulator:
    def __init__(self, mc_engine):
        self.mc_engine = mc_engine
        
    def initialize_growth_seed(self, tiling_data, seed_center, radius=5.0):
        """Initialize growth using EXISTING growth_status field"""
        # Reset all tiles to "ungrown"
        for tile in tiling_data["tiles"]:
            tile["growth_status"] = "ungrown"
        
        # Set seed region
        seed_tiles = self._find_tiles_in_radius(seed_center, radius, tiling_data)
        for tile in seed_tiles:
            tile["growth_status"] = "seed"
        
        self._update_growth_frontier(tiling_data)
        return tiling_data
    
    def propagate_growth_front(self, tiling_data, obstacles):
        """
        Grow one step with LOCAL healing constraints
        FIX: Run MC on full tiling but restrict to growth region
        """
        # 1. Save original flip eligibility
        original_eligibility = self._save_flip_eligibility(tiling_data)
        
        try:
            # 2. Restrict flips to growth region only
            self._restrict_flips_to_growth_region(tiling_data)
            
            # 3. Run MC healing on FULL tiling (but only growth region flips)
            self.mc_engine.run_mc_sweep(tiling_data, steps=100)
            
            # 4. Add new growth tiles
            new_tiles = self._add_growth_layer(tiling_data, obstacles)
            
        finally:
            # 5. RESTORE original flip eligibility (critical!)
            self._restore_flip_eligibility(tiling_data, original_eligibility)
        
        return tiling_data
    
    def _restrict_flips_to_growth_region(self, tiling_data) -> None:
        """
        FIX: Modify flip eligibility to only allow growth region flips
        This affects LegalFlipEngine._is_eligible_for_flip behavior
        """
        for tile in tiling_data["tiles"]:
            # Only allow flips in growth frontier and recently grown regions
            if tile["growth_status"] in ["frontier", "recently_grown"]:
                tile["_growth_flippable"] = True
            else:
                tile["_growth_flippable"] = False
    
    def _save_flip_eligibility(self, tiling_data) -> Dict[int, bool]:
        """Save original flip eligibility state"""
        return {tile["id"]: tile.get("flippable", True) for tile in tiling_data["tiles"]}
    
    def _restore_flip_eligibility(self, tiling_data, original_eligibility: Dict) -> None:
        """Restore original flip eligibility"""
        for tile in tiling_data["tiles"]:
            tile["flippable"] = original_eligibility[tile["id"]]
            # Remove temporary growth restriction
            if "_growth_flippable" in tile:
                del tile["_growth_flippable"]
```

```python
class TileSignatureValidator:
    """
    FIX: Properly initialized with classifier instance
    Validates tile-centered local environments
    """
    
    def __init__(self, initial_tiling):
        from src.energy.combinatorial_classifier import CombinatorialVertexClassifier
        self.classifier = CombinatorialVertexClassifier()  # FIX: Store instance
        self.allowed_signatures = self._build_tile_signature_catalog(initial_tiling)
    
    def _build_tile_signature_catalog(self, tiling_data) -> Set[str]:
        """Build catalog of allowed tile-centered signatures"""
        signatures = set()
        
        for tile_id in range(len(tiling_data["tiles"])):
            signature = self._compute_tile_signature(tile_id, tiling_data)
            signatures.add(signature)
        
        print(f"✅ Built tile signature catalog: {len(signatures)} allowed patterns")
        return signatures
    
    def _compute_tile_signature(self, tile_id: int, tiling_data) -> str:
        """Compute rotation-normalized tile signature"""
        environment = self.classifier._analyze_vertex_environment(tile_id, tiling_data)
        
        # Create canonical signature: "CENTER_TYPE-NEIGHBOR_TYPES-ANGLES"
        neighbor_types = "-".join(sorted(environment.neighbor_tile_types))
        angle_hash = self._hash_angle_sequence(environment.angle_sequence)
        
        return f"{environment.center_tile_type}-{neighbor_types}-{angle_hash}"
    
    def validate_flip(self, cluster_ids: List[int], tiling_data) -> Tuple[bool, str]:
        """
        Validate flip preserves quasiperiodicity
        Returns: (is_valid, message)
        """
        # Get all tiles affected by flip (cluster + neighbors)
        affected_tiles = self._get_flip_neighborhood(cluster_ids, tiling_data)
        
        for tile_id in affected_tiles:
            signature = self._compute_tile_signature(tile_id, tiling_data)
            if signature not in self.allowed_signatures:
                return False, f"Illegal tile pattern at tile {tile_id}: {signature}"
        
        return True, "Flip preserves quasiperiodicity"
```
